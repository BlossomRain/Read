\documentclass[border=10pt,crop=true]{standalone}
\usepackage{circuitikz}
\usetikzlibrary{calc, arrows.meta}
\input{02-op-circuit-style.tex}

% ---------------------------------------------------------------------------
% Left panel: Designer default op amp (− top, + bottom) — input low, R bus high
% ---------------------------------------------------------------------------
\def\OpCy{1.05}           % op-amp center y
\def\RfY{2.35}            % top feedback bus (v_- → R → v_o)
\def\SepX{4.55}
\def\SepGap{0.11}
\def\SepH{2.55}
\def\RightX{5.55}
\def\GndY{0}
\def\ViSrcDx{0.65}
\def\ViSrcDrop{1.15}
% --- right model ---
\def\BusY{1.45}
\def\RxVi{0.9}
\def\RxRi{2.05}
\def\RxIsrc{3.55}
\def\RxRoSpan{1.6}
\def\RxGndW{8.2}

\begin{document}
\begin{circuitikz}[american, scale=1.0, transform shape]

  % ========================================================================
  % LEFT — v-i converter
  %   − on top (R feedback bus); + on bottom (v_i); no input/feedback crossing
  % ========================================================================
  \begin{scope}
    \node[opfigThin, anchor=south] at (1.85, 2.95) {v--i converter};

    \coordinate (OAC) at (2.35, \OpCy);
    \node[op amp](OA) at (OAC) {};

    \coordinate (VM) at (0.65, \RfY);
    \coordinate (FBR) at (2.55, \RfY);
    \coordinate (OutTap) at (2.55, \OpCy);
    \coordinate (Io) at (3.55, \OpCy);

    % v_- : VM (top bus) -- vertical down -- OA.-
    \draw (VM) -- (VM |- OA.-);
    \draw (VM |- OA.-) -- (OA.-);
    \node[circ, fill=black, inner sep=0.75pt] at (VM) {};

    % R on top bus: VM → output  (Designer: 8.31,6.99 -- R -- 10.81,6.99 |- out)
    \draw (VM) to[R] (FBR);
    \node[opfigThin] at ($ (VM)!0.5!(FBR)+(0,4mm) $) {$R$};
    \draw (FBR) -- (OutTap) -- (OA.out);
    \draw (OA.out) to[short, -o] (Io);
    \node[above=1mm, opfig] at (Io) {$i_o$};

    % v_i -> (+) on bottom  (Designer: 7.75,4.75 -| 8.31,4.76; Vsrc to gnd)
    \coordinate (VplusTap) at ($(OA.+)+(-0.2, 0)$);
    \coordinate (VsrcTop) at ($(VplusTap)+(-\ViSrcDx, 0)$);
    \coordinate (VsrcBot) at ($(VsrcTop)+(0, -\ViSrcDrop)$);
    \draw (OA.+) -- (VplusTap);
    \draw (VplusTap) -| (VsrcTop);
    \draw (VsrcTop) to[V] (VsrcBot);
    \node[ground, scale=0.85] at (VsrcBot) {};
    \node[opfigThin, anchor=east] at ($ (VsrcTop)!0.5!(VsrcBot)+(-5mm,0) $) {$v_i$};
  \end{scope}

  % ========================================================================
  % MIDDLE — divider
  % ========================================================================
  \begin{scope}[xshift=\SepX cm]
    \coordinate (SepBot) at (0, \GndY-0.35);
    \coordinate (SepTop) at (0, \GndY-0.35+\SepH);
    \draw (SepBot) -- (SepTop);
    \draw ([xshift=\SepGap cm]SepBot) -- ([xshift=\SepGap cm]SepTop);
    \node[opfigThin, anchor=south west] at (0, 2.35) {model};
  \end{scope}

  % ========================================================================
  % RIGHT — port model
  % ========================================================================
  \begin{scope}[xshift=\RightX cm]
    \coordinate (Vbot) at (\RxVi, \GndY);
    \coordinate (Vtop) at (\RxVi, \BusY);
    \coordinate (Ritop) at (\RxRi, \BusY);
    \coordinate (Ribot) at (\RxRi, \GndY);
    \coordinate (Isbot) at (\RxIsrc, \GndY);
    \coordinate (Istop) at (\RxIsrc, \BusY);
    \coordinate (RoL) at (\RxIsrc+\RxRoSpan, \BusY);
    \coordinate (RoR) at (\RxIsrc+2*\RxRoSpan, \BusY);
    \coordinate (IoM) at (RoR);
    \coordinate (GndR) at (\RxIsrc+2*\RxRoSpan, \GndY);
    \coordinate (GndEnd) at (\RxGndW, \GndY);

    \draw (Vbot) -- (GndEnd);
    \node[ground, scale=0.85] at ($(Vbot)!0.22!(GndEnd)$) {};

    \draw (Vbot) to[V] (Vtop);
    \node[opfigThin, anchor=east] at ($ (Vbot)!0.5!(Vtop)+(-5mm,0) $) {$v_i$};
    \draw (Vtop) -- (Ritop);
    \draw (Ritop) to[R] (Ribot);
    \node[opfigThin, anchor=east] at ($ (Ritop)!0.5!(Ribot)+(-5mm,0) $) {$R_i$};

    \draw (Isbot) to[I] (Istop);
    \node[opfigThin, anchor=east] at ($ (Isbot)!0.5!(Istop)+(-5mm,0) $) {$G v_i$};
    \draw (Istop) -- (RoL);
    \draw (RoL) to[R] (RoR);
    \node[opfigThin] at ($ (RoL)!0.5!(RoR)+(0,4.5mm) $) {$R_o$};

    \draw (RoR) -- (IoM);
    \draw (IoM) to[short, -o] ++(0.55, 0);
    \node[above=1mm, opfig] at (IoM) {$i_o$};
    \draw (IoM) to[R] (GndR);
    \node[opfigThin, anchor=west] at ($ (IoM)!0.5!(GndR)+(5mm,0) $) {$R_L$};
  \end{scope}

\end{circuitikz}
\end{document}
