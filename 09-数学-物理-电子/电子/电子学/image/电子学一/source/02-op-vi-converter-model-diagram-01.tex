\documentclass[border=10pt,crop=true]{standalone}
\usepackage{circuitikz}
\usetikzlibrary{calc}

\ctikzset{
  bipoles/length=0.9cm,
  bipoles/thickness=1,
  label distance=2.5mm,
}
\tikzset{
  opfig/.style={font=\small},
  titlefig/.style={font=\small\bfseries},
}

% ---------------------------------------------------------------------------
% Layout — absolute X (cm), orthogonal wires
% ---------------------------------------------------------------------------
\def\BusY{1.45}
\def\GndY{0}
\def\LeftShiftX{1.15}    % left panel (clear of || at SepX)
\def\LeftShiftY{1.0}
\def\SepX{3.85}          % double line (right of i_o terminal)
\def\SepGap{0.11}
\def\SepH{2.55}
\def\RxOff{0.75}           % model block shift right (cm)
\def\RxVi{4.0}
\def\RxRi{5.2}
\def\RxIsrc{6.7}
\def\RxRo{8.0}
\def\RxRl{9.3}
\def\RxOut{10.15}
\def\RxGndEnd{10.6}

\begin{document}
\begin{circuitikz}[american, scale=1.0, transform shape]

  % ========================================================================
  % LEFT — v-i converter
  % ========================================================================
  \begin{scope}[shift={(\LeftShiftX, \LeftShiftY)}]
    \node[titlefig, anchor=south] at (0, 2.35) {v-i converter};

    \node[op amp, noinv input down] (opamp) at (0, 0) {};

    % (-) -> left -> up: vertical R (offset left) -> ground above R
    \draw (opamp.-) -- ++(-0.55, 0) coordinate (Rfork);
    \draw (Rfork) -- ++(0, 0.45) coordinate (Rbot);
    \draw (Rbot) to[R=$R$] ++(0, 0.55) coordinate (Rtop);
    \node[ground, scale=0.85, anchor=east] (gndR) at ($(Rtop)+(-0.55,0)$) {};
    \draw (gndR) -- (Rtop);

    % (+) left, down -> v_i terminal
    \draw (opamp.+) -- ++(-0.85, 0) coordinate (Vmid);
    \draw (Vmid) -- ++(0, -0.65) coordinate (Vstub);
    \draw (Vstub) to[short, o-] ++(-0.25, 0)
      node[left=1.5mm, opfig]{$v_i$};

    % i_o (shorter — stays left of ||)
    \draw (opamp.out) to[short, -o, i=$i_o$] ++(0.85, 0);
  \end{scope}

  % ========================================================================
  % MIDDLE — ||
  % ========================================================================
  \begin{scope}[xshift=\SepX cm]
    \coordinate (SepBot) at (0, \GndY-0.35);
    \coordinate (SepTop) at (0, \GndY-0.35+\SepH);
    \draw (SepBot) -- (SepTop);
    \draw ([xshift=\SepGap cm]SepBot) -- ([xshift=\SepGap cm]SepTop);
  \end{scope}

  % ========================================================================
  % RIGHT — model (shifted right by RxOff)
  % ========================================================================
  \begin{scope}[xshift=\RxOff cm]
  \node[titlefig, anchor=south west] at (\RxVi, 2.35) {model};

  \coordinate (Vbot) at (\RxVi, \GndY);
  \coordinate (Vtop) at (\RxVi, \BusY);
  \coordinate (Ritop) at (\RxRi, \BusY);
  \coordinate (Ribot) at (\RxRi, \GndY);
  \coordinate (Isbot) at (\RxIsrc, \GndY);
  \coordinate (Istop) at (\RxIsrc, \BusY);
  \coordinate (RoTop) at (\RxRo, \BusY);
  \coordinate (RoBot) at (\RxRo, \GndY);
  \coordinate (RlTop) at (\RxRl, \BusY);
  \coordinate (RlBot) at (\RxRl, \GndY);
  \coordinate (OutN) at (\RxOut, \BusY);
  \coordinate (GndEnd) at (\RxGndEnd, \GndY);

  \draw (Vbot) -- (GndEnd);
  \node[ground, scale=0.85] at ($(Vbot)!0.22!(GndEnd)$) {};

  % input loop only: v_i || R_i (top tie Vtop--Ritop; no tie to output loop)
  \draw (Vbot) to[V, v=$v_i$] (Vtop);
  \draw (Ritop) to[R=$R_i$] (Ribot);
  \draw (Vtop) -- (Ritop);

  % output loop only: G v_i -> R_o -> R_L -> i_o (top bus NOT connected to R_i)
  \draw (Isbot) to[I, l_=$G v_i$] (Istop);
  \draw (Istop) -- (RoTop) -- (RlTop) -- (OutN);
  \draw (RoTop) to[R=$R_o$] (RoBot);
  \draw (RlTop) to[R=$R_L$] (RlBot);

  \draw (OutN) to[short, -o, i=$i_o$] ++(0.4, 0);
  \end{scope}

\end{circuitikz}
\end{document}
