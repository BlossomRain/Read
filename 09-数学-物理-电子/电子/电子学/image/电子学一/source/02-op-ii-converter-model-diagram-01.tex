% ---------------------------------------------------------------------------
% I–I converter — left panel: − top / + bottom (§2.8.3 / §2.7.2 style)
%   R2 input; top-bus feedback wire v_o → v_- (no R1 symbol)
% ---------------------------------------------------------------------------
\def\OpCy{1.05}
\def\RfY{2.35}
\def\SepX{4.55}
\def\SepGap{0.11}
\def\SepH{2.55}
\def\RightX{5.55}
\def\GndY{0}
\def\RxInBase{0.9}
\def\RxInL{-0.55}
\def\RxIsrcX{3.55}
\def\RxRoSpan{1.6}
\def\RxGndW{8.2}

\begin{document}
\begin{circuitikz}[american, scale=1.0, transform shape]

  % ========================================================================
  % LEFT — i-i converter
  %   − on top (i_i via R2); + on bottom (ground); feedback wire on top bus
  % ========================================================================
  \begin{scope}
    \node[opfigThin, anchor=south] at (1.85, 2.95) {i--i converter};

    \coordinate (OAC) at (2.35, \OpCy);
    \node[op amp](OA) at (OAC) {};

    \coordinate (VM) at (0.65, \RfY);
    \coordinate (FBR) at (2.55, \RfY);
    \coordinate (OutTap) at (2.55, \OpCy);
    \coordinate (Io) at (3.55, \OpCy);

    % v_- : VM (top bus) -- vertical down -- OA.-
    \draw (VM) -- (VM |- OA.-);
    \draw (VM |- OA.-) -- (OA.-);
    \node[circ, fill=black, inner sep=0.75pt] at (VM) {};

    % i_i -- R2 -- v_-
    \draw ($(VM)+(-1.35,0)$) node[left, opfig]{$i_i$} to[short, o-] ($(VM)+(-1.05,0)$);
    \draw ($(VM)+(-1.05,0)$) to[R] (VM);
    \node[opfigThin, anchor=south] at ($ (VM)+(-0.7,0)+(0,-4mm) $) {$R_2$};

    % feedback wire on top bus: v_- → v_o (R1 omitted; sets α in text)
    \draw (VM) -- (FBR);
    \draw (FBR) -- (OutTap) -- (OA.out);
    \draw (OA.out) to[short, -o] (Io);
    \node[above=1mm, opfig] at (Io) {$i_o$};

    % v_+ grounded on bottom
    \draw (OA.+) -- ++(0, -0.95) node[ground, scale=0.85]{};
  \end{scope}

  % ========================================================================
  % MIDDLE — divider
  % ========================================================================
  \begin{scope}[xshift=\SepX cm]
    \coordinate (SepBot) at (0, \GndY-0.35);
    \coordinate (SepTop) at (0, \GndY-0.35+\SepH);
    \draw (SepBot) -- (SepTop);
    \draw ([xshift=\SepGap cm]SepBot) -- ([xshift=\SepGap cm]SepTop);
    \node[opfigThin, anchor=south west] at (0, 2.35) {model};
  \end{scope}

  % ========================================================================
  % RIGHT — port model (current out; tops disconnected)
  % ========================================================================
  \begin{scope}[xshift=\RightX cm]
    \coordinate (InL) at (\RxInBase+\RxInL, \RfY);
    \coordinate (In) at (\RxInBase, \RfY);
    \coordinate (GndIn) at (\RxInBase, \GndY);
    \coordinate (Isbot) at (\RxIsrcX, \GndY);
    \coordinate (Istop) at (\RxIsrcX, \RfY);
    \coordinate (RoL) at (\RxIsrcX+\RxRoSpan, \RfY);
    \coordinate (RoR) at (\RxIsrcX+2*\RxRoSpan, \RfY);
    \coordinate (IoM) at (RoR);
    \coordinate (GndR) at (\RxIsrcX+2*\RxRoSpan, \GndY);
    \coordinate (GndEnd) at (\RxGndW, \GndY);

    \draw (GndIn) -- (GndEnd);
    \node[ground, scale=0.85] at ($(GndIn)!0.22!(GndEnd)$) {};

    \draw (InL) to[short, o-] (In);
    \node[above=1mm, opfig] at ($ (InL)!0.5!(In) $) {$i_i$};
    \draw (In) to[R] (GndIn);
    \node[opfigThin, anchor=east] at ($ (In)!0.5!(GndIn)+(-5mm,0) $) {$R_i$};

    \draw (Isbot) to[I] (Istop);
    \node[opfigThin, anchor=east] at ($ (Isbot)!0.5!(Istop)+(-5mm,0) $) {$\alpha i_i$};
    \draw (Istop) -- (RoL);
    \draw (RoL) to[R] (RoR);
    \node[opfigThin] at ($ (RoL)!0.5!(RoR)+(0,4.5mm) $) {$R_o$};

    \draw (RoR) -- (IoM);
    \draw (IoM) to[short, -o] ++(0.55, 0);
    \node[above=1mm, opfig] at (IoM) {$i_o$};
    \draw (IoM) to[R] (GndR);
    \node[opfigThin, anchor=west] at ($ (IoM)!0.5!(GndR)+(5mm,0) $) {$R_L$};
  \end{scope}

\end{circuitikz}
\end{document}
