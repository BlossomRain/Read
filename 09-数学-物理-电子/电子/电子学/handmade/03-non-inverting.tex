% Layout from noninverting_amp.tikz — top-bus feedback, fixed junctions
\documentclass[border=10pt,crop=true]{standalone}
\usepackage{circuitikz}
\usetikzlibrary{calc}

\ctikzset{label distance=3mm}

\begin{document}
\begin{circuitikz}[american, scale=1.0, transform shape]
  \node[op amp] (OA) at (7.75, 5.75) {};
  \coordinate (VM) at (5.25, 6.25);
  \coordinate (FBUS) at (5.25, 7.5);
  \coordinate (OUTJ) at (9, 7.5);

  % v_i -> (+)
  \draw (4.25, 5.25) node[left, font=\small]{$v_i$} to[short, o-] (5.75, 5.25);
  \draw (5.75, 5.25) -| (OA.+);

  % (-) junction: R1 to ground, stub to top bus
  \draw (OA.-) -| (VM);
  \node[circ, fill=black, inner sep=0.9pt] at (VM) {};
  \draw (VM) to[american resistor, l^=$R_1$] (3.5, 6.25);
  \draw (3.5, 6.25) -- (2.5, 6.25) node[ground, scale=0.85]{};

  % R2: VM -> top bus -> R2 -> output tap -> v_o
  \draw (VM) -- (FBUS);
  \draw (FBUS) -- (6.5, 7.5) to[american resistor, l^=$R_2$] (7.75, 7.5)
    -- (OUTJ) coordinate(OUTTAP);
  \draw (OUTTAP) |- (OA.out);
  \draw (OA.out) -- (10.5, 5.75) node[right, font=\small]{$v_o$};
\end{circuitikz}
\end{document}
